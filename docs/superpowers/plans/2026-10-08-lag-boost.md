> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Живой раннер счёта rsi_pullback умножает долю входа rsi_pullback/rsi_zone на `LAG_BOOST_MULT`, если тикер за `LAG_BOOST_DAYS` завершённых дней отстал от IMOEX на `LAG_BOOST_PCT` п.п. и больше; опция включается через ENV.

**Architecture:** Чистое решение — пакет `livecore/lagboost` (без I/O). Данные индекса: UID IMOEX через новый метод gRPC-клиента `IndicativeUIDByTicker` (RPC `Indicatives`), дневные закрытия — новая функция `marketdata.IndexDailyCloses`. Точка применения — `buy()` в `rsi_pullback/live/pass.go` перед `sizing.Lots`; индекс запрашивается лениво, один раз за пасс, и только при включённой опции. Бэктест и `cmd/pullparity` не трогаются.

**Tech Stack:** Go 1.25, gRPC (Tinkoff Invest API), heetch/confita, testify + mockery v2, mage.

**Spec:** `docs/superpowers/specs/2026-10-08-lag-boost-design.md`

## Global Constraints

- Ветка: `feat/lag-boost` (уже создана, спека закоммичена `da15c4f7`). На `main` не коммитить, не пушить, не деплоить.

- Бюджет по-прежнему ограничен свободным кэшем (`sizing.Lots` не меняется).
- Комментарии и тексты уведомлений — по-русски, как в окружающем коде; идентификаторы — английские.

## Review Focus

1. Сегодняшняя (незавершённая) дневная свеча IMOEX попала в ряд — заглядывание: lag считался бы по живой цене индекса против вчерашней цены тикера. Ожидание: свечи с `Time` не раньше MSK-полуночи `now` и `IsComplete == false` отброшены (тест в Task 4).

```go
// Package lagboost решает, увеличить ли долю входа по тикеру, который за последние
// Days завершённых дней отстал от индекса IMOEX на Pct п.п. и больше. Пакет чистый:
// ряды закрытий собирает раннер (без сегодняшней незавершённой свечи), здесь — только
// арифметика. Обоснование и пороги — docs/superpowers/specs/2026-10-08-lag-boost-design.md.
package lagboost

import (
	"errors"
	"fmt"
	"strconv"
)

// Config — настройки LAG_BOOST_* (общие для раннера счёта rsi_pullback).
type Config struct {
	Enabled bool
	Mult    float64 // множитель доли входа
	Pct     float64 // порог отставания, п.п.
	Days    int     // окно в завершённых днях
}

// Validate проверяет границы только у включённой опции: выключенная с любыми числами
// безопасна. Верхняя граница множителя — защита от опечатки в ENV, а не торговое решение.
func (c Config) Validate() error {
	if !c.Enabled {
		return nil
	}
	var errs []error
	if c.Mult < 1 || c.Mult > 3 {
		errs = append(errs, fmt.Errorf("LAG_BOOST_MULT=%v вне [1, 3]", c.Mult))
	}
	if c.Pct <= 0 {
		errs = append(errs, fmt.Errorf("LAG_BOOST_PCT=%v должен быть > 0", c.Pct))
	}
	if c.Days < 1 || c.Days > 60 {
		errs = append(errs, fmt.Errorf("LAG_BOOST_DAYS=%d вне [1, 60]", c.Days))
	}
	return errors.Join(errs...)
}

// Decision — итог для одного входа. Mult всегда ≥ 1; Applied — сработал ли буст.
type Decision struct {
	Mult    float64
	Lag     float64 // доходность тикера минус доходность индекса за окно, п.п.
	Applied bool
	Reason  string // почему буст не применён (для лога); пусто, если применён
}

// Decide сравнивает доходности за последние Days+1 закрытий каждого ряда (сдвиг считается
// по барам своего ряда). Любая нехватка данных даёт множитель 1: буст необязателен.
func Decide(tickerCloses, indexCloses []float64, c Config) Decision {
	if !c.Enabled {
		return Decision{Mult: 1, Reason: "выключено"}
	}
	n := c.Days
	if len(tickerCloses) < n+1 || len(indexCloses) < n+1 {
		return Decision{Mult: 1, Reason: fmt.Sprintf("мало истории: тикер %d, индекс %d закрытий, нужно %d",
			len(tickerCloses), len(indexCloses), n+1)}
	}
	t0, t1 := tickerCloses[len(tickerCloses)-1-n], tickerCloses[len(tickerCloses)-1]
	x0, x1 := indexCloses[len(indexCloses)-1-n], indexCloses[len(indexCloses)-1]
	if t0 <= 0 || t1 <= 0 || x0 <= 0 || x1 <= 0 {
		return Decision{Mult: 1, Reason: "нулевая или отрицательная цена в окне"}
	}
	lag := (t1/t0-1)*100 - (x1/x0-1)*100
	if lag <= -c.Pct {
		return Decision{Mult: c.Mult, Lag: lag, Applied: true}
	}
	return Decision{Mult: 1, Lag: lag, Reason: fmt.Sprintf("отставание %.2f п.п. не дотягивает до %v", -lag, c.Pct)}
}

// Note — строка для уведомления о входе; пустая, если буст не применён.
func (d Decision) Note(days int) string {
	if !d.Applied {
		return ""
	}
	return fmt.Sprintf("Размер ×%s: отстал от IMOEX на %.1f%% за %dд",
		strconv.FormatFloat(d.Mult, 'f', -1, 64), -d.Lag, days)
}
```

```go
func (c *instrumentsServiceClient) IndicativeUIDByTicker(ctx context.Context, ticker string) (string, error) {
```

```go
// mskZone — граница «сегодня» для дневного ряда индекса: та же MSK-полночь, по которой
// visibleDaily режет дневки тикера в AssembleMarketData.
var mskZone = func() *time.Location {
	loc, err := time.LoadLocation("Europe/Moscow")
	if err != nil {
		return time.FixedZone("MSK", 3*3600)
	}
	return loc
}()

// IndexDailyCloses — закрытия завершённых дневных свечей индекса (IMOEX для буста доли)
// до MSK-полуночи now, oldest-first. Окно запроса — days·2+20 календарных дней, не меньше
// 30: хватает на days торговых дней с праздниками. Сегодняшняя свеча отбрасывается даже с
// IsComplete — иначе отставание тикера сравнивалось бы с живым индексом.
func IndexDailyCloses(ctx context.Context, c candles.CandleClient, uid string, days int,
	now time.Time) ([]float64, error) {

	span := max(days*2+20, 30)
	from := now.AddDate(0, 0, -span)
	raw, err := c.GetCandles(ctx, &uid, enum.Day1.ToNumberInvestAPI(),
		timestamppb.New(from), timestamppb.New(now), nil, true)
	if err != nil {
		return nil, fmt.Errorf("rsi_pullback marketdata: index daily candles: %w", err)
	}
	cs := candles.ToCandles(raw, true)
	sort.Slice(cs, func(a, b int) bool { return cs[a].Time.Before(cs[b].Time) })
	n := now.In(mskZone)
	bound := time.Date(n.Year(), n.Month(), n.Day(), 0, 0, 0, 0, mskZone)
	out := make([]float64, 0, len(cs))
	for _, cd := range cs {
		if cd.Time.Before(bound) {
			out = append(out, cd.Close)
		}
	}
	return out, nil
}
```

```go
// lagBoost решает множитель доли для входа. Любой сбой — множитель 1 и WARN: буст
// необязателен и вход не блокирует.
func (s *service) lagBoost(ctx context.Context, pc *passCtx, sl *slot, ticker string,
	md strategy.MarketData) lagboost.Decision {

	if !s.lag.Enabled || !lagBoostStrategies[sl.strat.Name()] {
		return lagboost.Decision{Mult: 1}
	}
	if !pc.indexLoaded {
		pc.indexLoaded = true
		pc.index, pc.indexErr = s.loadIndex(ctx, pc)
		if pc.indexErr != nil {
			logger.Warn(fmt.Sprintf("rsi_pullback: LAG_BOOST — индекс недоступен, вход без буста: %v", pc.indexErr))
		}
	}
	if pc.indexErr != nil {
		return lagboost.Decision{Mult: 1}
	}
	d := lagboost.Decide(md.DailyCloses, pc.index, s.lag)
	if !d.Applied {
		logger.InfoContext(ctx, fmt.Sprintf("%s: %s LAG_BOOST не применён: %s", sl.strat.Name(), ticker, d.Reason))
	}
	return d
}

func (s *service) loadIndex(ctx context.Context, pc *passCtx) ([]float64, error) {
	if s.imoexUID == "" {
		uid, err := s.instruments.IndicativeUIDByTicker(ctx, imoexTicker)
		if err != nil {
			return nil, err
		}
		s.imoexUID = uid
	}
	return marketdata.IndexDailyCloses(ctx, s.market, s.imoexUID, s.lag.Days, pc.now)
}
```

```markdown
### Буст доли: `LAG_BOOST_*`

Общая опция раннера счёта. Перед входом rsi_pullback или rsi_zone раннер сравнивает доходность
тикера и индекса IMOEX за `LAG_BOOST_DAYS` последних завершённых дней (сегодняшняя свеча не
участвует). Если тикер отстал на `LAG_BOOST_PCT` п.п. и больше, доля входа умножается на
`LAG_BOOST_MULT`; бюджет по-прежнему ограничен свободным кэшем. Сигналы и выходы не меняются —
только число лотов. gap_fade буст не получает.

| Переменная | Дефолт | Смысл |
|---|---|---|
| `LAG_BOOST_ENABLED` | `false` | включение |
| `LAG_BOOST_MULT` | `2` | множитель доли, допустимо 1–3 |
| `LAG_BOOST_PCT` | `5` | порог отставания, п.п., > 0 |
| `LAG_BOOST_DAYS` | `5` | окно в завершённых днях, 1–60 |

Неверные числа выключают опцию с ERROR при старте. UID индекса берётся из справочника
`Indicatives`, дневные свечи IMOEX — не чаще раза за пасс и только при сигнале на вход. Сбой
индекса или короткая история дают множитель 1 и WARN; вход не блокируется. Уведомление о
входе с бустом содержит строку `Размер ×2: отстал от IMOEX на N% за 5д`, а если кэш урезал
лоты — пометку `(урезано кэшем)`. Бэктест и `cmd/pullparity` буст не моделируют: он не меняет
решений, а эффект виден только на уровне общего счёта.
```
